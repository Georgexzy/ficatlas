# FicAtlas — crawler policy
#
# This matters more here than on a typical site, for two reasons.
#
# 1. Search URLs are an infinite space. Every combination of q, fandoms, tags,
#    ratings, sort and page is a distinct URL backed by a query over 19.7M rows.
#    A crawler that follows filter links will generate unbounded work against a
#    database running on a home connection. That is the classic crawler trap and
#    it is the single most likely way this site falls over.
#
# 2. The reader serves other people's fanfiction in full. Those authors posted
#    to AO3 and FFN, not here, and did not agree to their work appearing under
#    this domain in Google results. Keeping /story/*/chapter/* out of the index
#    is a courtesy that costs nothing — people who want to read arrive through
#    search on this site, not through a search engine landing mid-chapter.
#
# Story detail pages are allowed: they are metadata plus a link to the original,
# which is the part worth being findable and is also what makes the index useful
# to someone who has not heard of it.

User-agent: *

# Private and app-only routes.
Disallow: /account
Disallow: /library
Disallow: /login
Disallow: /settings
Disallow: /api/
# Owner-only. Gated server-side, but there is no reason for it to be crawled.
Disallow: /admin

# Full text of works hosted here. See note 2 above.
Disallow: /story/*/chapter/

# The infinite space. `/?q=` and friends are search results, not content.
Disallow: /*?
Allow: /$
Allow: /story/

# Fandom hubs — the only route a crawler has into the index.
#
# Story pages were allowed above long before anything linked to them: search
# results are the sole path to a work, they live at `/?q=` URLs, and those are
# blocked directly above. So /story/ was permitted and unreachable at the same
# time, which is the state this fixes.
#
# The hubs are a bounded alternative to a sitemap of all 19.9M works: a few
# thousand pages, each listing a capped set of works, rather than an invitation
# to crawl the whole index off a home connection. See backend/fandom_hubs.py.
Allow: /fandoms
Allow: /fandom/
Allow: /sitemap.xml

# Politeness. Google ignores this; most other crawlers honour it, and those are
# the ones most likely to crawl without restraint.
Crawl-delay: 10

# Training crawlers get nothing. The index is assembled from other archives, and
# the works in it are not ours to hand over in bulk — the same position the OTW
# takes for AO3.
#
# This list was six agents and had drifted. Cloudflare Radar's May 2026 figures
# put Amazonbot at 3.23% of AI crawler traffic, Meta-ExternalAgent at 3.22% and
# Applebot-Extended at 2.98% — all larger than several agents already named here,
# and none of them listed. A blocklist that names GPTBot and misses Amazonbot is
# not the position it claims to be.
#
# Two things worth knowing about this list rather than treating it as magic:
#
#   * Google-Extended and Applebot-Extended are not crawlers. They are opt-out
#     tokens with nothing behind them — blocking them does not stop a fetch, it
#     changes what Google and Apple may do with pages their ordinary crawlers
#     already have. Included for that reason, not as traffic control.
#   * Bytespider and Meta's agents are widely reported not to honour robots.txt
#     reliably. Nothing here enforces anything; it is a request. If that matters,
#     it has to be an edge rule — Cloudflare has a one-click AI-scraper block,
#     and the tunnel in docker-compose.prod.yml is where that would apply.
User-agent: GPTBot
Disallow: /

User-agent: CCBot
Disallow: /

User-agent: ClaudeBot
Disallow: /

User-agent: anthropic-ai
Disallow: /

User-agent: Bytespider
Disallow: /

User-agent: Amazonbot
Disallow: /

User-agent: Meta-ExternalAgent
Disallow: /

User-agent: Meta-ExternalFetcher
Disallow: /

User-agent: PerplexityBot
Disallow: /

User-agent: Applebot-Extended
Disallow: /

User-agent: Google-Extended
Disallow: /

User-agent: cohere-ai
Disallow: /

User-agent: Diffbot
Disallow: /

User-agent: Omgilibot
Disallow: /

User-agent: AI2Bot
Disallow: /

# Retrieval agents, which fetch a page because a person asked a question rather
# than to train on it. Blocking these is a genuinely different decision from the
# list above: it costs discovery, since it is how the site would be cited in an
# AI answer. They are blocked anyway, because AO3 blocks ChatGPT-User and this
# file claims to hold the OTW's position — and because the works being fetched
# are not ours to serve into someone else's product, whatever the purpose.
User-agent: ChatGPT-User
Disallow: /

User-agent: OAI-SearchBot
Disallow: /

User-agent: Claude-User
Disallow: /

User-agent: Claude-SearchBot
Disallow: /

User-agent: Perplexity-User
Disallow: /

# is allowed above; crawlers find it without being told. Add the line below,
# with the real host, once a domain exists:
